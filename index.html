    <script>

        const questions = [

            "回想初次相识，你对我最深刻的 3 个直觉印象是什么？",

            "在你认识的所有人里，你觉得我身上哪种特质是最罕见、最难以取代的？",

            "你觉得我在谈论什么话题时，眼神里最有“光”？",

            "你觉得我的存在，更像是一团“温暖的火”，还是一个“深邃的湖”？",

            "如果我突然去创业，你会建议我做“掌舵者”还是“压舱石”？",

            "当我陷入低谷时，你觉得我最需要的支持方式是什么？",

            "你觉得我在社交场合中展示出的“距离感”通常是因为什么？",

            "如果我是一本还没写完的书，你觉得下章是“破浪前行”还是“大隐于市”？",

            "在你眼中，我更倾向于“直觉驱动”还是“逻辑至上”？",

            "有没有哪种才华，是你觉得我拥有但还没被大家看到的？",

            "如果要给我提一个能让我变得更好的“真诚差评”，你会说什么？",

            "在什么样的情况下，你会毫不犹豫地把一件大事交给我？",

            "如果必须用一个词来定义我的“精神底色”，你会选哪个词？",

            "最后，有没有一句话是你想送给现在或未来的我的？"

        ];


        let current = 1;

        const container = document.getElementById('step-container');

        const scriptURL = '你的APPS_SCRIPT_URL_贴在这里';


        // 动态生成题目

        questions.forEach((q, i) => {

            const stepId = i + 2;

            const div = document.createElement('div');

            div.id = `step-${stepId}`;

            div.className = "step hidden fade-in";

            div.innerHTML = `

                <label class="block text-xs font-bold text-indigo-400 uppercase tracking-widest mb-2">Question ${stepId}/15</label>

                <h2 class="text-xl font-bold text-slate-800 mb-6">${q}</h2>

                <textarea name="q${stepId}" rows="4" class="w-full bg-slate-50 border-none rounded-2xl p-4 focus:ring-2 focus:ring-indigo-500 outline-none transition" placeholder="输入你的回答..."></textarea>

                <button type="button" onclick="next(${stepId})" class="mt-8 w-full bg-indigo-600 text-white py-4 rounded-2xl font-bold">下一题</button>

            `;

            container.appendChild(div);

        });


        function next(stepId) {

            const currentEl = document.getElementById(`step-${stepId}`);

            const input = currentEl.querySelector('input, textarea');

            if (input && !input.value.trim()) {

                alert("留点笔墨吧，这对我很重要。");

                return;

            }

            

            currentEl.classList.add('hidden');

            current++;

            const nextEl = document.getElementById(`step-${current}`) || document.getElementById('step-final');

            nextEl.classList.remove('hidden');

            document.getElementById('progress').style.width = (current / 16 * 100) + '%';

        }


        document.getElementById('survey-form').addEventListener('submit', e => {

            e.preventDefault();

            const btn = document.getElementById('submit-btn');

            btn.disabled = true;

            btn.innerText = "提交中...";

            

            fetch(scriptURL, { method: 'POST', body: new FormData(e.target)})

            .then(res => {

                alert('提交成功！谢谢你！');

                btn.innerText = "已完成";

            })

            .catch(err => alert('出错啦，请截图联系我：' + err.message));

        });
